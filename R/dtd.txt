get_gene_annotation_bygeneid <- function(geneids = c()){
  loadCaGenomeAnnotation()
  if (length(geneids) == 0) {
    return(CaGenomeAnnotation)
  } else {
    gene_annotation <- CaGenomeAnnotation[CaGenomeAnnotation$geneid %in% geneids, ]
    return(gene_annotation)
  }
}

get_geneids_bygoids<- function(goids){
  loadCaProteinEnrichment()
  geneids<-CaProteinEnrichment[CaProteinEnrichment$termid %in% goids,]$geneid
  geneids<-unique(geneids)
  geneids<-geneids[!is.na(geneids)]
  return(geneids)
}

get_gene_enrichment_bygeneid<-function(geneids){
  loadCaProteinEnrichment()
  gene_enrichment<-CaProteinEnrichment[CaProteinEnrichment$geneid %in% geneids,]
  return(gene_enrichment)
}

get_goids_bykewords <- function(keywords){
  loadCaProteinEnrichment()
  terms<-CaProteinEnrichment$term
  terms<-tolower(terms)
  keywords<-tolower(keywords)
  goids<-CaProteinEnrichment[grepl(paste(keywords, collapse="|"), terms),]$termid
  goids<-unique(goids)
  return(goids)
}


get_geneids_bygotermkeywords<-function(keywords){
  goids<-get_goids_bykewords(keywords)
  geneids<-get_geneids_bygoids(goids)
  return(geneids)
}

get_geneids_bygoids<- function(goids){
  loadCaProteinEnrichment()
  geneids<-list()
  for (goid in goids){
    geneids[[goid]]<-CaProteinEnrichment[CaProteinEnrichment$termid==goid,]$geneid
  }
  return(geneids)
}

get_study_data <- function(study = "all") {
  if (is.vector(study)) {
    stop("Please provide a single study name")
  }
  data <- list()
  loadCaExpressionCounts()
  loadCaExpressionMetadata()
  if (study == "all") {
    data$counts <- CaExpressionCounts
    data$metadata <- CaExpressionMetadata
  } else {
    studysamples<-rownames(CaExpressionMetadata[CaExpressionMetadata$study == study, ])
    data$counts <- CaExpressionCounts[, colnames(CaExpressionCounts) %in% studysamples]
    data$metadata <- CaExpressionMetadata[CaExpressionMetadata$study == study, ]
  }
  return(data)
}

get_samples_data <- function(samples = c()) {
  data <- list()
  loadCaExpressionCounts()
  loadCaExpressionMetadata()

    data$counts <- CaExpressionCounts
    data$metadata <- CaExpressionMetadata
  } else {
    data$counts <- CaExpressionCounts[, colnames(CaExpressionCounts) %in% samples]
    data$metadata <- CaExpressionMetadata[rownames(CaExpressionMetadata) %in% samples, ]
  }
  return(data)
}


get_metadata <- function(samples = NULL, group = NULL, experiment = NULL, study = NULL) {
  if (sum(is.null(c(samples, group, experiment, study))) > 1) {
    stop("Please provide only one argument")
  }
  loadCaExpressionMetadata()
  metadata <- CaExpressionMetadata
  if (!is.null(samples)) {
    metadata <- metadata[rownames(metadata) %in% samples, ]
  }
  if (!is.null(group)) {
    metadata <- metadata[metadata$group == group, ]
  }
  if (!is.null(experiment)) {
    metadata <- metadata[metadata$Experiment == experiment, ]
  }
  if (!is.null(study)) {
    metadata <- metadata[metadata$study == study, ]
  }
  return(metadata)
}

get_gwas_data <- function(geneids, pvalue = 0.05, phenotype = NULL) {
  loadCaGWAS()
  gwas_data <- CaGWAS[CaGWAS$geneid %in% geneids & CaGWAS$pvalue < pvalue, ]
  if (!is.null(phenotype)) {
    gwas_data <- gwas_data[gwas_data$phenotype == phenotype, ]
  }
  return(gwas_data)
}
get_gene_express_bygeneid <- function(geneids, gene_express_count=NULL, study=NULL, experiment=NULL){
  geneExpressionCounts <- NULL
  if (is.null(gene_express_count)) {
    print("Gene expression count data is not provided. Using default data.")
    loadCaExpressionCounts()
    geneExpressionCounts <- CaExpressionCounts
  } else {
    geneExpressionCounts <- gene_express_count
  }
  gene_express <- geneExpressionCounts[rownames(geneExpressionCounts) %in% geneids, ]
  if (!is.null(study)) {
    loadCaExpressionMetadata()
    studysamples<-rownames(CaExpressionMetadata[CaExpressionMetadata$study == study, ])
    gene_express <- gene_express[, colnames(gene_express) %in% studysamples]
  }
  if (!is.null(experiment)) {
    loadCaExpressionMetadata()
    experimentsamples<-rownames(CaExpressionMetadata[CaExpressionMetadata$experiment == experiment, ])
    gene_express <- gene_express[, colnames(gene_express) %in% experimentsamples]
  }
  return(gene_express)
}

filter_gene_expression<-function(data, min_expression=10){
  data[is.na(data)]<-0
  data<-data[rowSums(data)>min_expression,]
  return(data)
}

get_gene_logfc<-function(genesSample, pvalue=1, fdr=1, study="all", experiment="all"){
  loadCaExpressionSignificance()
  geneExp<-CaExpressionSignificance[CaExpressionSignificance$geneid %in% genesSample,]
  if (pvalue != 1) {
    geneExp<-geneExp[geneExp$pvalue<=pvalue,]
    print("some genes may have been removed due to pvalue")
  }
  if (fdr != 1) {
    geneExp<-geneExp[geneExp$fdr<=fdr,]
    print("some genes may have been removed due to fdr")
  }
  if (study != "all") {
    geneExp<-geneExp[geneExp$study==study,]
  }
  if (experiment != "all") {
    geneExp<-geneExp[geneExp$experiment==experiment,]
  }
  print("some genes may have several logfc values, the mean is taken")
  geneExp<-aggregate(logfc~geneid, geneExp, mean)
  return(geneExp)
}

get_gene_logfc_bycategory<-function(geneids, pvalue=1, fdr=1, category="study"){
  loadCaExpressionSignificance()
  geneExp<-CaExpressionSignificance[CaExpressionSignificance$geneid %in% geneids,]
  if (pvalue != 1) {
    geneExp<-geneExp[geneExp$pvalue<=pvalue,]
    print("some genes may have been removed due to pvalue")
  }
  if (fdr != 1) {
    geneExp<-geneExp[geneExp$fdr<=fdr,]
    print("some genes may have been removed due to fdr")
  }
  if (category == "study") {
    geneExp<-aggregate(logfc~geneid+study, geneExp, mean)
  }
  if (category == "experiment") {
    geneExp<-aggregate(logfc~geneid+experiment, geneExp, mean)
  }
  return(geneExp)
}

get_protein_interaction_table <- function(genes, only_interacting=TRUE) {
  loadCaProteinInteractionNetwork()
  if (only_interacting) {
    network<-CaProteinInteractionNetwork[CaProteinInteractionNetwork$p1_geneid%in%genes & CaProteinInteractionNetwork$p2_geneid%in%genes,]
  } else {
    network<-CaProteinInteractionNetwork[CaProteinInteractionNetwork$p1_geneid%in%genes | CaProteinInteractionNetwork$p2_geneid%in%genes,]
  }
  return(network)
}

map_expression_to_color <- function(expr, min_expr, max_expr) {
  negative_colors <- colorRampPalette(c('blue', 'white'))(10)
  positive_colors <- colorRampPalette(c('yellow', 'red'))(10)
  
  if (expr < 0) {
    bin <- cut(expr, breaks = seq(min_expr, 0, length.out = 6), include.lowest = TRUE, labels = FALSE)
    return(negative_colors[bin])
  } else {
    bin <- cut(expr, breaks = seq(0, max_expr, length.out = 11), include.lowest = TRUE, labels = FALSE)
    return(positive_colors[bin])
  }
}


plot_protein_interaction_network_with_gene_expression <- function(network_table, geneExp, targetCol="score", title="Protein Interaction Network", file=NULL, plot_width = 10, plot_height=10, minnodes=3) {
  edges <- network_table[, c("p1_geneid", "p2_geneid", targetCol)]
  colnames(geneExp)<-c("geneid","expression")
  g <- graph_from_data_frame(d = edges, directed = FALSE)
  if (minnodes > 0) {
    g<-delete_vertices(g, V(g)[degree(g) < minnodes])
    print(paste("Note: Removed nodes with less than", minnodes, "connections"))
  }
  min_expr <- min(geneExp$expression, na.rm = TRUE)
  max_expr <- max(geneExp$expression, na.rm = TRUE)
  
  unmatched_genes <- setdiff(V(g)$name, geneExp$geneid)
  if (length(unmatched_genes) > 0) {
    print(paste("Warning: The following genes are missing expression data:", paste(unmatched_genes, collapse = ", ")))
  }
  
  V(g)$color <- sapply(V(g)$name, function(gene) {
    expr <- geneExp$expression[geneExp$geneid == gene]
    if (length(expr) > 0 && !is.na(expr)) {
      map_expression_to_color(expr, min_expr, max_expr)
    } else {
    }
  })
  
  
  E(g)$color <- apply(colorRamp(c('blue', 'yellow', 'red'))((E(g)$score - min(E(g)$score)) / (max(E(g)$score) - min(E(g)$score))), 1, function(x) rgb(x[1]/255, x[2]/255, x[3]/255))
  
  layout <- layout_with_fr(g, weights = edge_lengths)
  
  plot(g, layout = layout, vertex.color = V(g)$color, edge.color = E(g)$color, 
       vertex.size = 10, edge.width = 2)
  
  title(main = title)
  
  mtext(paste("Edge lengths represent", targetCol), side = 1, line = 3, cex = 0.8, col="red")
  
  if (!is.null(file)) {
    dev.copy(pdf, file, width = plot_width, height = plot_height)
    dev.off()
  }
}

generate_heatmap_by_go_term_keyword<-function(keyword, study=NULL, experiment=NULL, outfolder="", plotwidth=10, plotheight=10){
  genes<-get_geneids_bygotermkeywords(keyword)
  gene_annotation<-get_gene_annotation_bygeneid(unlist(genes))
  gene_enrichment<-get_gene_enrichment_bygeneid(unlist(genes))
  genesExpress<-get_gene_express_bygeneid(unlist(genes), study=study, experiment=experiment)
  genesExpress<-filter_gene_expression(genesExpress, min_expression=10)
  metadata<-get_metadata(colnames(genesExpress))
  metadata<-metadata[,c("group","study","experiment")]
  pheatmap(as.matrix(genesExpress), scale="row",annotation_col=metadata,
           col=colorRampPalette(c("blue", "yellow", "red"))(256))
  if (outfolder!="") {
    if (!file.exists(outfolder)) {
      dir.create(outfolder)
    }
    dev.copy(pdf, paste(outfolder, paste0(keyword, "_heatmap.pdf"), sep="/"), width=plotwidth, height=plotheight)
    dev.off()
    write.csv(gene_annotation, paste(outfolder, paste0(keyword, "_annotation",".csv"), sep="/"))
    write.csv(gene_enrichment, paste(outfolder, paste0(keyword, "_enrichment",".csv"), sep="/"))
  }
}

gene_pheno_GWASmatrix_byGOKeyword<-function(keyword, pvalue=0.05,phenotype=NULL){
  genes<-get_geneids_bygotermkeywords(keyword)
  genes<-unlist(genes)
  GWASdata<-get_gwas_data(genes,pvalue = pvalue, phenotype = phenotype)
  gene_pheno_matrix<-reshape2::dcast(GWASdata, geneid~phenotype, value.var="pvalue")
  return(gene_pheno_matrix)
}

align_gene_expression_metadata<-function(dataList = NULL, metadata = NULL, counts = NULL){
  gene_meta <- list()
  if (!is.null(dataList)) {
    if (length(dataList) == 0) {
      stop("Please provide a list of gene expression data")
    }
    counts <- dataList$counts
    metadata <- dataList$metadata
  }
  
  if (!is.null(metadata) && !is.null(counts)) {
    if (length(metadata) == 0) {
      stop("Please provide metadata")
    }
    if (length(counts) == 0) {
      stop("Please provide counts")
    }
  }
  
  if (is.null(metadata) || is.null(counts)) {
    stop("Both metadata and counts must be provided.")
  }
  
  counts <- counts[, colnames(counts) %in% rownames(metadata)]
  
  metadata <- metadata[rownames(metadata) %in% colnames(counts),]
  
  gene_meta$counts <- counts
  gene_meta$metadata <- metadata
  
  return(gene_meta)
}

get_data_report <- function(data) {
  report <- list()
  report$samples <- ncol(data$counts)
  report$genes <- nrow(data$counts)
  report$studies <- length(unique(data$metadata$study))
  report$experiments <- length(unique(data$metadata$experiment))
  print("These data have been generated from the ChickpeaOmicsR package.")
  print(paste("The data has", report$samples, "samples and", report$genes, "genes."))
  print(paste("The data is from", report$studies, "studies", "and", report$experiments, "experiments."))
  return(report)
}
